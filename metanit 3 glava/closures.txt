package main

import "fmt"

func outer() func() {
	var n int = 99
	inner := func() {

		n *= 2
		fmt.Println(n)
	}
	return inner
}

func multiply(n int) func(int) int {
	return func(m int) int { return n * m }
}

func main() {

	fn := outer()
	fn()
	fn()
	fn()
	
    fn2 := multiply(9)
	result := fn2(3)
	fmt.Println(result)
}
