package main

import "fmt"

func main() {
	var e int = 4
	fmt.Printf("a = %b\n", e)
	e = e << 3
	fmt.Printf("a = %b\n", e)
	var a int = 10
	b := 5

	a += b
	fmt.Println("a += b:", a)

	a -= b
	fmt.Println("a -= b:", a)

	a *= 10
	fmt.Println("a *= 10:", a)

	a /= b
	fmt.Println("a /= b:", a)

	a %= 15
	fmt.Println("a %= 15:", a)

	a <<= 2
	fmt.Println("a <<= 2:", a)

	a >>= 1
	fmt.Println("a >>= 1:", a)

	a &= 8
	fmt.Println("a &= 8:", a)

	a ^= 10
	fmt.Println("a ^= 10:", a)

	a |= 5
	fmt.Println("a |= 5:", a)
}
