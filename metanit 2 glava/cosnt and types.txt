package main

import (
	"fmt"
)

func main() {

	var age = 41
	var isEnabled = false
	var message = "Hello"
	var koef = 1.2

	fmt.Printf("age type: %T\n", age)
	fmt.Printf("isEnabled type: %T\n", isEnabled)
	fmt.Printf("message type: %T\n", message)
	fmt.Printf("koef type: %T\n", koef)

	const pi float64 = 3.1415
	fmt.Println(pi)

	const (
		a = 2
		b
		c
		d = 3
		e
		f = 4
	)
	fmt.Println(a, b, c, d, e, f)

	const (
		C0 = iota
		C1
		C2
		C3 = iota
	)
	fmt.Println(C0, C1, C2, C3)

	const (
		C4 = iota
	)
	fmt.Println(C4)

}
