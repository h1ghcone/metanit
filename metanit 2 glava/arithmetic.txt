package main

import "fmt"

func main() {
	var a float32 = 4
	var b float32 = 6
	var c = a + b
	var e = a - b
	var d = a * b
	var f float32 = a / b
	fmt.Println(c, e, d, f)
	var h int = 1
	var g int = 100
	h++
	g--
	fmt.Println(h, g)
    
    var l int = 6
	var m int = 8
	var v bool = l < m
	fmt.Println(v)
	var p bool = 3 < 4 && 3 > 4
	fmt.Println(p)
	var u bool = 3 < 4 || 3 > 4
	fmt.Println(u)
}
}
