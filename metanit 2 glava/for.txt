package main

import "fmt"

func main() {
	var i = 1
	//можно объявить переменную в цикле и вне цикла
	for ; i < 10; i++ {
		fmt.Println(i * i)
	}

	var a = 1
	for a < 10 {
		fmt.Println(a * a)
		a++
	}

	for b := 1; b < 10; b++ {
		for c := 1; c < 10; c++ {
			fmt.Print(b*c, "\t")
		}
		fmt.Println()
	}

	str := "welcome"
	for index, value := range str {
		fmt.Printf("Index: %d, Value: %c\n", index, value)
	}

	var peoples = [2]string{"Michael", "Bob"}
	for index, value := range peoples {
		fmt.Println(index, value)
	}
OuterLoop:
	for i := 1; i <= 3; i++ {
		for j := 1; j <= 3; j++ {

			fmt.Printf("i = %d, j = %d\n", i, j)

			if i == 2 && j == 2 {

				fmt.Println("Выход из внешнего цикла...")

				break OuterLoop
			}
		}
	}

	fmt.Println("Цикл завершен...")
}
